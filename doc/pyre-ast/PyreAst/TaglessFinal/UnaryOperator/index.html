Module TaglessFinal.UnaryOperator

This module provides a type that represents Python unary operators.

type 'a t = private {
  1. invert : 'a;
    (*

    Represents Python operator ~.

    *)
  2. not_ : 'a;
    (*

    Represents Python operator not.

    *)
  3. uadd : 'a;
    (*

    Represents Python operator + (unary).

    *)
  4. usub : 'a;
    (*

    Represents Python operator - (unary).

    *)
}
